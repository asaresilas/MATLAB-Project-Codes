PART 2.2: Prediction Interface - MATLAB Function Block
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

☐ Step 2.2.1: Create predict_maintenance function block
   [SIMULINK FUNCTION BLOCK REQUIRED] *** CRITICAL ***
   
   Block type: MATLAB Function (Simulink Editor → Library → MATLAB Function)
   Name: predict_maintenance
   
   INPUTS:
   - sensor_buffer: (1024×1) buffered sensor data
   
   OUTPUTS:
   - pred_probability: (1×1) failure probability 0-1
   - alert_level_out: (1×1) numeric: 0=GREEN, 1=YELLOW, 2=RED
   - confidence_out: (1×1) confidence score
   - latency_ms: (1×1) inference time in milliseconds
   
   Function body (paste into block):
   ═════════════════════════════════════════════════════════════────────
   function [pred_probability, alert_level_out, confidence_out, latency_ms] = ...
       predict_maintenance(sensor_buffer)
       
       persistent client connected initialized
       
       % One-time initialization
       if isempty(initialized)
           client = PredictiveMaintenanceClient('ws://localhost:8002', 'MOTOR-TEST-001');
           connected = 0;
           initialized = true;
       end
       
       % Try to connect on first call if not connected
       if ~connected
           try
               client.connect();
               connected = 1;
           catch
               connected = 0;
           end
       end
       
       % Default outputs if not connected
       pred_probability = 0.5;
       alert_level_out = 0;      % GREEN (safe default)
       confidence_out = 0.0;
       latency_ms = 0.0;
       
       % If connected, send prediction request
       if connected
           try
               % Create sensor names (example)
               sensor_names = {'VibX','VibY','VibZ','CurrA','CurrB',...
                               'CurrC','TempB','Speed','VibXF','VibYF',...
                               'VibZF','Pressure','TempW','TempA','CurrRMS'};
               
               % Send prediction
               t_start = tic;
               pred = client.predict(sensor_buffer', sensor_names);
               latency_ms = toc(t_start) * 1000;
               
               % Extract results
               pred_probability = pred.prediction;
               confidence_out = pred.confidence;
               
               % Map alert level string to numeric
               switch pred.alert_level
                   case 'CRITICAL'
                       alert_level_out = 2;  % RED
                   case 'WARNING'
                       alert_level_out = 1;  % YELLOW
                   otherwise
                       alert_level_out = 0;  % GREEN
               end
               
           catch ME
               % Graceful error handling
               warning(['Prediction failed: ' ME.message]);
               pred_probability = 0.5;
               alert_level_out = 0;
               confidence_out = 0.0;
           end
       end
   end
   ═════════════════════════════════════════════════════════════════════

   Code:
   ```matlab
   function log_ground_truth(fault_detected, time_sim, failure_type)
       persistent last_fault_time fault_sent
       
       if isempty(last_fault_time)
           last_fault_time = 0;
           fault_sent = false;
       end
       
       if fault_detected && ~fault_sent
           % Calculate days to failure (normalized: 1 day = 10 sim seconds)
           days_to_failure = (60 - time_sim) / 10;
           
           % Create client if not exists
           persistent client
           if isempty(client)
               client = PredictiveMaintenanceClient('ws://localhost:8002');
               client.connect();
           end
           
           % Send ground truth
           try
               client.sendGroundTruth(1, days_to_failure, failure_type);
               disp('[GROUND TRUTH] Sent to API');
               fault_sent = true;
           catch ME
               warning(ME.message);
           end
       end
   end
   ```
